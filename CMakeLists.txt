cmake_minimum_required(VERSION 3.22)
project(Singlr VERSION 0.1.0 LANGUAGES CXX)

find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Gui Qml Quick DBus Widgets)
find_package(KF6WindowSystem 6.0 REQUIRED)
find_package(KF6GlobalAccel 6.0 REQUIRED)
find_package(KF6NetworkManagerQt 6.0 REQUIRED)
find_package(KF6Wallet 6.0 REQUIRED)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBNM REQUIRED IMPORTED_TARGET libnm)

add_subdirectory(shell/session)
add_subdirectory(shell/command)
include(cmake/InstallDesktop.cmake)
include(cmake/InstallShellPackage.cmake)

option(SINGLR_BUILD_LIQUIDGLASS "Build the KWin 6.8.1 material effect" ON)
if(SINGLR_BUILD_LIQUIDGLASS)
    add_subdirectory(kwin/effects/singlr-liquidglass)
endif()
